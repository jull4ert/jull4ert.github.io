匯入器

using TicketDesk.Web.Models;

namespace TicketDesk.Web.Services;

public record ImportResult(List<Ticket> Tickets, List<string> Errors);

/// <summary>
/// 讀舊系統匯出的 CSV。原則：壞資料要回報（含行號與原因），不要猜、也不要默默丟掉。
/// 限制：這是簡化版，不處理欄位內含逗號或換行（真實匯出檔要用正式的 CSV 函式庫）。
/// </summary>
public static class LegacyTicketImporter
{
    public static ImportResult Parse(IEnumerable<string> lines)
    {
        var tickets = new List<Ticket>();
        var errors = new List<string>();
        var lineNo = 0;

        foreach (var line in lines)
        {
            lineNo++;
            if (lineNo == 1 || string.IsNullOrWhiteSpace(line))
            {
                continue; // 標題列與空行
            }

            var cols = line.Split(',');
            if (cols.Length != 5)
            {
                errors.Add($"第 {lineNo} 行：欄位數是 {cols.Length}，應為 5");
                continue;
            }

            if (cols[0].Length < 2 || !int.TryParse(cols[0][1..], out var id))
            {
                errors.Add($"第 {lineNo} 行：單號格式不符「{cols[0]}」");
                continue;
            }

            if (!RocDate.TryParse(cols[4], out var date))
            {
                errors.Add($"第 {lineNo} 行（{cols[0]}）：建立日期無法判斷「{cols[4]}」");
                continue;
            }

            tickets.Add(new Ticket
            {
                Id = id,
                Customer = cols[1],
                Subject = cols[2],
                Body = cols[3],
                CreatedAt = date.ToDateTime(TimeOnly.MinValue)
            });
        }

        return new ImportResult(tickets, errors);
    }
}

測試

// ---- Lab 6：舊系統匯出檔，壞資料要回報而不是猜 ----
public class LegacyImporterTests
{
    private static string[] Sample() => File.ReadAllLines(Path.Combine(AppContext.BaseDirectory, "docs", "tickets-export.csv"));

    [Fact]
    public void Imports_valid_rows_and_reports_invalid_ones_with_line_numbers()
    {
        var result = LegacyTicketImporter.Parse(Sample());
        Assert.Equal(4, result.Tickets.Count);
        Assert.Equal(5, result.Errors.Count);
        Assert.Contains(result.Errors, e => e.Contains("L004"));
        Assert.Contains(result.Errors, e => e.Contains("L005"));
    }

    [Fact]
    public void Converts_roc_dates_to_gregorian()
    {
        var result = LegacyTicketImporter.Parse(Sample());
        var first = result.Tickets.First(t => t.Id == 1);
        Assert.Equal(new DateTime(2024, 11, 5), first.CreatedAt.Date);
    }

    [Fact]
    public void Rows_with_wrong_column_count_are_reported()
    {
        var result = LegacyTicketImporter.Parse(new[] { "單號,客戶,主旨,內容,建立日期", "L100,甲,乙" });
        Assert.Empty(result.Tickets);
        Assert.Single(result.Errors);
    }
}

L007 的取捨

  • 目前的 RocDate 刻意只接受 2 到 3 位數的年份，所以「2024/11/08」被拒絕。這是保守的選擇：寧可請人確認，也不要猜。
  • 若客戶確認「四位數年份一定是西元」，可以擴充 RocDate 支援，並加上測試與說明；這個規則應由客戶確認，而不是你假設。
  • 更重要的是向客戶回報「這份匯出檔混用了兩種年份格式」，這是系統端的資料問題。
